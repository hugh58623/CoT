from typing import List, Optional
def normalize_scores(values: List[Optional[float]]) -> List[float]:
    vals = [v for v in values if v is not None]
    if not vals:
        return []
    mn, mx = min(vals), max(vals)
    if mn == mx:
        return []
    return [(x - mn) / (mx - mn) for x in vals]
